--- QuizService.java ---
@Service @Transactional @RequiredArgsConstructor
Inject : QuizRepository, QuizAttemptRepository, EnrollmentRepository,
         ProgressRepository, RedisTemplate<String,String>, CertificateService

private static final String QUIZ_SESSION_KEY = "quiz:session:%s:%s";

QuizResponseDto getQuizDefinition(UUID courseId, UUID userId)
  → quiz = quizRepository.findByCourseId(courseId) ou throw ResourceNotFoundException
  → enrollment check
  → int attemptsUsed = quizAttemptRepository.countByUserIdAndQuizId(userId, quiz.getId())
  → int remaining = quiz.getMaxAttempts() - attemptsUsed
  → si remaining <= 0 : throw ValidationException("Nombre maximum de tentatives atteint")
  → QuizResponseDto dto = map quiz (SANS correctAnswer)
  → dto.setRemainingAttempts(remaining)
  → return dto

void startQuiz(UUID userId, UUID quizId)
  → quiz = quizRepository.findById(quizId) ou throw ResourceNotFoundException
  → int used = quizAttemptRepository.countByUserIdAndQuizId(userId, quizId)
  → si used >= quiz.getMaxAttempts() : throw ValidationException("Plus de tentatives disponibles")
  → String key = QUIZ_SESSION_KEY.formatted(userId, quizId)
  → si redisTemplate.hasKey(key) : throw ValidationException("Session quiz déjà active")
  → redisTemplate.opsForValue().set(key, "ACTIVE", quiz.getTimerSeconds(), TimeUnit.SECONDS)

QuizResultDto submitQuiz(UUID userId, UUID quizId, QuizSubmitDto dto)
  → String key = QUIZ_SESSION_KEY.formatted(userId, quizId)
  → boolean sessionActive = redisTemplate.hasKey(key)
  → int score = 0
  → si sessionActive :
       redisTemplate.delete(key)
       score = calculateScore(quizId, dto.getAnswers())
  → (si !sessionActive : score = 0)
  → quiz = quizRepository.findById(quizId)...
  → boolean passed = score >= quiz.getPassScore()
  → QuizAttempt attempt = new QuizAttempt(userId, quizId, dto.getAnswers(), score, passed, now())
  → quizAttemptRepository.save(attempt)
  → si passed : certificateService.checkAndGenerate(userId, quiz.getCourse().getFormation().getId())
  → return QuizResultDto(score, passed, corrections, attempt.getId(), remainingAttempts)

private int calculateScore(UUID quizId, Map<String,Object> answers)
  → charger toutes les questions avec correctAnswer
  → pour chaque question : comparer answers.get(questionId) avec correctAnswer
  → totalPoints = somme des points si correct
  → return (int) ((double) earnedPoints / totalPoints * 100)

QuizResultDto getAttemptResults(UUID quizId, UUID attemptId, UUID userId)
  → attempt = quizAttemptRepository.findById(attemptId) ou throw ResourceNotFoundException
  → si attempt.getUserId() != userId : throw AccessDeniedException
  → charger questions avec correctAnswers
  → construire corrections : List<CorrectionDto(questionId, userAnswer, correctAnswer, isCorrect)>
  → return QuizResultDto complet avec corrections

@Service @Transactional @RequiredArgsConstructor
Inject : AttendanceRepository, RedisTemplate<String,String>

private static final String QR_KEY = "attendance:qr:%s";

String generateQrCodeToken(UUID seanceId)
  → String token = UUID.randomUUID().toString()
  → String key = QR_KEY.formatted(token)
  → redisTemplate.opsForValue().set(key, seanceId.toString(), 300, TimeUnit.SECONDS)
  → return token

void scanQrCode(UUID userId, String token)
  → String key = QR_KEY.formatted(token)
  → String seanceIdStr = redisTemplate.opsForValue().get(key)
  → si seanceIdStr == null : throw ResourceNotFoundException("QR Code expiré ou invalide")
  → UUID seanceId = UUID.fromString(seanceIdStr)
  → si attendanceRepository.existsByUserIdAndSeanceId(userId, seanceId) :
       throw ValidationException("Présence déjà enregistrée")
  → attendanceRepository.save(new Attendance(userId, seanceId, token, LocalDateTime.now()))
  → redisTemplate.delete(key)

@Service @RequiredArgsConstructor
Inject : ProgressRepository, QuizAttemptRepository, CourseRepository,
         MinioService, NotificationService, CertificateRepository

void checkAndGenerate(UUID userId, UUID formationId)
  → courses = courseRepository.findByFormationId(formationId)
  → pour chaque cours :
       si !progressRepository.existsCompletedByUserIdAndCourseId(userId, cours.getId()) : return
       si !quizAttemptRepository.existsPassedByUserIdAndQuizCourseId(userId, cours.getId()) : return
  → generate(userId, formationId)

String generate(UUID userId, UUID formationId)
  → User user = userRepository.findById(userId)...
  → Formation formation = formationRepository.findById(formationId)...
  → double avgScore = quizAttemptRepository.avgPassedScoreByUserAndFormation(userId, formationId)
  
  → PdfWriter writer = new PdfWriter(outputStream)
  → PdfDocument pdf = new PdfDocument(writer)
  → Document document = new Document(pdf, PageSize.A4.rotate())
  
  → document.add(new Paragraph("CERTIFICAT DE RÉUSSITE")
       .setFont(boldFont).setFontSize(32).setTextAlignment(CENTER))
  → document.add(new Paragraph("Décerné à").setFontSize(16))
  → document.add(new Paragraph(user.getFirstName() + " " + user.getLastName())
       .setFont(boldFont).setFontSize(24))
  → document.add(new Paragraph("Pour avoir complété avec succès")
  → document.add(new Paragraph(formation.getTitle()).setFont(boldFont).setFontSize(20))
  → document.add(new Paragraph("Score moyen : " + String.format("%.1f%%", avgScore)))
  → document.add(new Paragraph("Délivré le : " + LocalDate.now().format(DateTimeFormatter.ofPattern("dd MMMM yyyy", Locale.FRENCH))))
  
  → String verificationCode = UUID.randomUUID().toString()
  → BarcodeQRCode qrCode = new BarcodeQRCode("https://api.elearning.com/verify/" + verificationCode)
  → Image qrImage = new Image(qrCode.createFormXObject(pdf)).setWidth(80).setHeight(80)
  → document.add(qrImage)
  → document.close()
  
  → String objectKey = "certificates/" + userId + "/" + formationId + ".pdf"
  → minioService.uploadBytes(outputStream.toByteArray(), "elearning-media", objectKey)
  → certificateRepository.save(new Certificate(userId, formationId, objectKey, verificationCode, now()))
  → notificationService.sendToUser(userId, "Certificat disponible",
       "Votre certificat pour \"" + formation.getTitle() + "\" est prêt",
       Map.of("type", "CERTIFICATE", "formationId", formationId.toString()))
  → return objectKey
