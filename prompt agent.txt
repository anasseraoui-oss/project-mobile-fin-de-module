```java
@Component
public class SecurityUtils {
    public static UUID getCurrentUserId() {
        var auth = SecurityContextHolder.getContext().getAuthentication();
        var jwt = (Jwt) auth.getPrincipal();
        return UUID.fromString(jwt.getSubject());
    }

    public static String getCurrentUserRole() {
        var auth = SecurityContextHolder.getContext().getAuthentication();
        return auth.getAuthorities().stream()
            .findFirst()
            .map(GrantedAuthority::getAuthority)
            .orElse("");
    }

    public static UUID getCurrentOrganisationId() {
        var auth = SecurityContextHolder.getContext().getAuthentication();
        var jwt = (Jwt) auth.getPrincipal();
        String orgId = jwt.getClaimAsString("organisationId");
        return orgId != null ? UUID.fromString(orgId) : null;
    }
}
```

  → int attemptsUsed = quizAttemptRepository.countByUserIdAndQuizId(userId, quiz.getId())

void startQuiz(UUID userId, UUID quizId)
  → quiz = quizRepository.findById(quizId) ou throw ResourceNotFoundException
  → int used = quizAttemptRepository.countByUserIdAndQuizId(userId, quizId)
  → si used >= quiz.getMaxAttempts() : throw ValidationException("Plus de tentatives disponibles")
  → String key = QUIZ_SESSION_KEY.formatted(userId, quizId)
  → si redisTemplate.hasKey(key) : throw ValidationException("Session quiz déjà active")
  → redisTemplate.opsForValue().set(key, "ACTIVE", quiz.getTimerSeconds(), TimeUnit.SECONDS)

QuizResultDto submitQuiz(UUID userId, UUID quizId, QuizSubmitDto dto)
  → String key = QUIZ_SESSION_KEY.formatted(userId, quizId)
  → boolean sessionActive = redisTemplate.hasKey(key)
  → int score = 0
  → si sessionActive :
       redisTemplate.delete(key)
       score = calculateScore(quizId, dto.getAnswers())
  → (si !sessionActive : score = 0)
  → quiz = quizRepository.findById(quizId)...
  → boolean passed = score >= quiz.getPassScore()
  → QuizAttempt attempt = new QuizAttempt(userId, quizId, dto.getAnswers(), score, passed, now())
  → quizAttemptRepository.save(attempt)
  → si passed : certificateService.checkAndGenerate(userId, quiz.getCourse().getFormation().getId())
  → return QuizResultDto(score, passed, corrections, attempt.getId(), remainingAttempts)
